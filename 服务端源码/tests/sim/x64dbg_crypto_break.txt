// x64dbg 断点脚本 —— 0x43cc50 流密码三元组抓取（方案 B）
// ==========================================================
// 目标：抓 (seed, 明文, 密文) 三元组，离线反推 VM 内密钥流算法
//
// 前置：
//   1. 客户端使用【原版】mxmain.dll（加密开启，勿用补丁版）
//   2. x64dbg 附加游戏进程（运行到登录界面再附加，或附加后重启到登录）
//   3. 断点全部设在非虚拟化代码；0x43cc50 本体在 VMProtect 区段内【不断】，
//      避免破坏 VM 字节码执行上下文（VMProtect 对断点敏感）
//
// 用法：
//   脚本标签页 → 加载本文件 → 运行（自动设 3 个入口断点）
//   客户端登录 → 手动记录下面列出的参数与调用前后缓冲区
//   完成后：日志标签页 → 复制全部 → 发回分析
//
// 说明：x64dbg 脚本表达式支持 mod.base("模块名")；
//       若你的 x64dbg 版本不支持，把 base 改成符号窗里 mxmain.dll 的加载基址。

var base
mov base, mod.base("mxmain.dll")

// ── ① SetSeed 入口（0x430a60, __thiscall, ECX=CNet 对象）────────────────
//    记录: seed = [esp+4] (uint32), this = ECX
//    静态反编译 FUN_00430a60/FUN_00430790 显示的字段槽:
//      +0x1088 / +0x108c (发送上下文/参数)
//      +0x2098 / +0x209c (接收上下文/参数)
//    这些是传给初始化/变换函数的字段，不应当作状态内存本身。
bp base+0x30a60

// ── ② 发送加密入口（0x430ae0, __thiscall）──────────────────────────────
//    入口记录: plain_ptr = [esp+4], len = [esp+8], this = ECX
//    FUN_00430ae0 先复制明文到新缓冲区，再调用 0x43cc50 原地变换后发送。
//    记录密文时，应在进入变换后、发送前读取“复制后的缓冲区”，不是 plain_ptr。
//    在入口断点逐步执行到 call 0x43cc50；对 call 使用 Step Over，再记变换后缓冲区。
bp base+0x30ae0

// ── ③ 收包解密入口（0x430b50, __thiscall）──────────────────────────────
//    入口记录: len = [esp+4], buf = [esp+8], this = ECX
//    在函数尾部调用 0x43cc50 前后分别记录 buf，区分 socket 原始字节与解密结果。
bp base+0x30b50

log "断点已设置（mxmain.dll 基址=%s）" base
log "  0x430a60 SetSeed    : seed=[esp+4]"
log "  0x430ae0 发送加密   : buf=[esp+4] len=[esp+8]  （call 0x43cc50 后 buf=密文）"
log "  0x430b50 收包解密   : len=[esp+4] buf=[esp+8]  （call 0x43cc50 后 buf=明文）"
